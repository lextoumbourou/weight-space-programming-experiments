## Step 5

The multiplicity fix gave a big jump (33%). Now the main distortion is purely a **prediction-frequency imbalance**: digit **8** is predicted 3336 times (true ≈1000), while **5** is predicted only 108 times. The per-class correlations have different baselines.

Since the only lever I have is the feedback, I'll use the `fc.bias` to rebalance: compute `bias_d = -k·ln(pred_count_d / 1000)` from this step's confusion-matrix column totals, nudging over-predicted classes (8, 1, 4) down and starved classes (5, 3, 2, 0) up. Everything else stays the same.

# column totals (predicted counts) observed at step 4
PRED = np.array([463, 1576, 341, 238, 1562, 108, 659, 792, 3336, 925], dtype=np.float32)


def build_weights():
    scale = 2.0

    conv1_w = np.zeros((8, 1, 5, 5), dtype=np.float32)
    for k in range(8):
        conv1_w[k, 0, 2, 2] = 1.0
    conv1_b = np.zeros((8,), dtype=np.float32)

    conv2_w = np.zeros((16, 8, 5, 5), dtype=np.float32)
    for c in range(16):
        conv2_w[c, 0, c // 4, c % 4] = 1.0
    conv2_b = np.zeros((16,), dtype=np.float32)

    T = {}
    for d, rows in TEMPLATES.items():
        g = np.array([[1.0 if ch == '#' else 0.0 for ch in r] for r in rows],
                     dtype=np.float32)
        g = g - g.mean()
        n = np.linalg.norm(g)
        T[d] = g / n if n > 0 else g

    cnt = np.zeros((12, 12), dtype=np.float32)
    for c in range(16):
        ro, co = c // 4, c % 4
        for u in range(4):
            for v in range(4):
                cnt[min(ro + 2 * u, 11), min(co + 2 * v, 11)] += 1.0
    cnt[cnt == 0] = 1.0

    fc_w = np.zeros((10, 256), dtype=np.float32)
    for d in range(10):
        for c in range(16):
            ro, co = c // 4, c % 4
            for u in range(4):
                for v in range(4):
                    idx = c * 16 + u * 4 + v
                    r = min(ro + 2 * u, 11)
                    col = min(co + 2 * v, 11)
                    fc_w[d, idx] = scale * T[d][r, col] / cnt[r, col]

    # rebalancing bias from observed prediction frequencies
    k_bias = 0.6
    fc_b = (-k_bias * np.log(PRED / 1000.0)).astype(np.float32)

    return {
        "conv1.weight": conv1_w, "conv1.bias": conv1_b,
        "conv2.weight": conv2_w, "conv2.bias": conv2_b,
        "fc.weight": fc_w, "fc.bias": fc_b,
    }
